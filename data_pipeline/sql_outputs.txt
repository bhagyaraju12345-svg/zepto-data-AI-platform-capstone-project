Q5_JOIN
======================================================================
SELECT
            b.title,
            c.category_name,
            b.price_gbp,
            b.price_inr,
            b.rating,
            b.in_stock
        FROM books AS b
        INNER JOIN categories AS c
            ON b.category_id = c.category_id
        ORDER BY
            b.rating DESC,
            c.category_name,
            b.title
        LIMIT 10
